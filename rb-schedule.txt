// RB Schedule -> Calendar feed. Save this in Excel as:  RB Schedule Sync
// Returns Gio's whole assigned schedule (every row with hours in the Gio column),
// from the current week back through all future dates, as calendar-ready events.
// NOTE: YEAR is fixed to the sheet name below; update both when the 2027 tab arrives.

interface Ev {
  subject: string;
  start: string;      // "YYYY-MM-DDTHH:mm:00" if timed, "YYYY-MM-DD" if all-day
  end: string;        // timed: +2h ; all-day: exclusive end (the day after the last day)
  isAllDay: boolean;
  location: string;
  body: string;
  uid: string;
}

function main(workbook: ExcelScript.Workbook): Ev[] {
  const SHEET = "RBW 2026";
  const YEAR = 2026;

  const sheet = workbook.getWorksheet(SHEET);
  if (!sheet) return [];
  const used = sheet.getUsedRange();
  if (!used) return [];
  const vals = used.getValues();

  // Locate columns from the repeated header row (positions are constant across the sheet).
  const want = ["Order #","Total SO Cost","Install Cost","Ship Date","Install Date",
                "Salesperson","Notes","Brandon","Adam","Manny","Gio","Mike","RB Truck","Assigned Crew"];
  const col: { [k: string]: number } = {};
  for (let r = 0; r < vals.length; r++) {
    let hits = 0;
    for (let c = 0; c < vals[r].length; c++) if (want.indexOf(String(vals[r][c]).trim()) >= 0) hits++;
    if (hits >= 4) for (let c = 0; c < vals[r].length; c++) {
      const t = String(vals[r][c]).trim();
      if (want.indexOf(t) >= 0 && col[t] === undefined) col[t] = c;
    }
    if (col["Salesperson"] !== undefined && col["Gio"] !== undefined && col["Order #"] !== undefined) break;
  }
  const cOrder = col["Order #"], cSales = col["Salesperson"], cNotes = col["Notes"], cGio = col["Gio"];

  const dateRe = /^\s*(\d{1,2})\/(\d{1,2})(?:\s*-\s*(\d{1,2})\/(\d{1,2}))?/;
  const timeRe = /(\d{1,2})(?::(\d{2}))?\s*([ap])\.?m\.?(?![a-z])/i;
  const pad = (n: number) => (n < 10 ? "0" + n : "" + n);
  const isoDate = (mo: number, da: number) => `${YEAR}-${pad(mo)}-${pad(da)}`;
  function addDay(iso: string): string {
    const p = iso.split("-").map(x => Number(x));
    const d = new Date(Date.UTC(p[0], p[1] - 1, p[2]));
    d.setUTCDate(d.getUTCDate() + 1);
    return `${d.getUTCFullYear()}-${pad(d.getUTCMonth() + 1)}-${pad(d.getUTCDate())}`;
  }

  // Keep the current week back plus everything forward; drop older rows.
  const now = new Date();
  const cutoff = new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate() - 7));

  const seen: { [k: string]: boolean } = {};
  const out: Ev[] = [];

  for (let r = 0; r < vals.length; r++) {
    const a = String(vals[r][0]).trim();
    const m = a.match(dateRe);
    if (!m) continue;

    const gioVal = cGio !== undefined ? String(vals[r][cGio]).trim() : "";
    if (gioVal === "" || gioVal === "0") continue;   // WHOLE assigned schedule = Gio has hours

    const mo = Number(m[1]), da = Number(m[2]);
    const emo = m[3] ? Number(m[3]) : mo, eda = m[4] ? Number(m[4]) : da;
    const dStart = isoDate(mo, da), dEnd = isoDate(emo, eda);
    if (new Date(dEnd + "T00:00:00Z") < cutoff) continue;   // drop old rows

    const multiDay = dStart !== dEnd;
    let title = a.replace(dateRe, "").replace(/^[-\s\/]+/, "").trim();

    const tm = a.match(timeRe);
    let start = "", end = "", isAllDay = true;
    if (!multiDay && tm) {
      let hh = Number(tm[1]); const mm = tm[2] ? Number(tm[2]) : 0;
      const ap = tm[3].toLowerCase();
      if (ap === "p" && hh < 12) hh += 12;
      if (ap === "a" && hh === 12) hh = 0;
      if (hh >= 0 && hh <= 23 && mm >= 0 && mm <= 59) {
        start = `${dStart}T${pad(hh)}:${pad(mm)}:00`;
        let eh = hh + 2; if (eh > 23) eh = 23;
        end = `${dStart}T${pad(eh)}:${pad(mm)}:00`;
        isAllDay = false;
      }
    }
    if (isAllDay) { start = dStart; end = addDay(dEnd); }

    if (tm) title = title.replace(tm[0], "");
    title = title.replace(/\(\s*\)/g, "").replace(/\s{2,}/g, " ").replace(/^[-\s\/]+|[-\s]+$/g, "").trim();
    if (!title) title = "RB job";

    const order = cOrder !== undefined ? String(vals[r][cOrder]).trim() : "";
    const sales = cSales !== undefined ? String(vals[r][cSales]).trim() : "";
    const notes = cNotes !== undefined ? String(vals[r][cNotes]).trim() : "";

    let location = "";
    if (notes && /\d{2,}\s+[A-Za-z]/.test(notes)) location = notes;   // lift a street address if present

    const bp: string[] = [];
    if (order) bp.push("Order: " + order);
    if (sales) bp.push("Salesperson: " + sales);
    bp.push("Gio hrs: " + gioVal);
    if (notes) bp.push("Notes: " + notes);
    const body = bp.join("\n");

    const uid = `${dStart}|${title}`.toLowerCase();
    if (seen[uid]) continue;   // dedup exact repeats
    seen[uid] = true;

    out.push({ subject: title, start, end, isAllDay, location, body, uid });
  }

  return out;
}
